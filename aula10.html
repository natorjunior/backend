Passo 1

Extraia o repository

1. repository2. service3. controller4. router5. app 

// repositories/tarefaRepository.js
const tarefas = [
  { id: 1, titulo: "Estudar módulos", concluida: false },
  { id: 2, titulo: "Criar API", concluida: true }
];

export function listar() {
  return tarefas;
}

export function buscarPorId(id) {
  return tarefas.find((tarefa) => tarefa.id === id);
}

export function salvar(tarefa) {
  tarefas.push(tarefa);
  return tarefa;
}
O array fica privado. Outras camadas acessam os dados pelas funções do repository.
Passo 2

Adicione o service

1. repository2. service3. controller4. router5. app 

// services/tarefaService.js
import * as repository
  from "../repositories/tarefaRepository.js";

export function listar() {
  return repository.listar();
}

export function buscarPorId(id) {
  return repository.buscarPorId(id);
}

export function criar(titulo) {
  const tarefa = { id: Date.now(), titulo, concluida: false };
  return repository.salvar(tarefa);
}
O service decide como criar a tarefa; o repository decide como salvá-la.
Passo 3

Adicione o controller

1. repository2. service3. controller4. router5. app 

// controllers/tarefaController.js
import * as tarefas from "../services/tarefaService.js";

export function listar(req, res) {
  return res.json(tarefas.listar());
}

export function criar(req, res) {
  const { titulo } = req.body;

  if (!titulo?.trim()) {
    return res.status(400).json({ erro: "Título obrigatório" });
  }

  const tarefa = tarefas.criar(titulo);
  return res.status(201).json(tarefa);
}

Token

Como a API sabe quem está fazendo a requisição?

Problema · 3 min

O usuário já fez login. E agora?

Imagine que o usuário acabou de entrar no sistema:

POST /login
{
"email": "[ana@email.com](mailto:ana@email.com)",
"senha": "1234"
}

Logo depois, ele faz outra requisição:

GET /perfil
Como a API sabe que essa requisição é da Ana?
Ideia · 4 min

É aí que entra o token

Login
email + senha
→
API
gera um token
→
Cliente
guarda o token 
{
"token": "8f2a91cd..."
}
O token passa a ser a identificação usada nas próximas requisições.
Implementação · 8 min

Vamos gerar um token com JWT

import express from "express";
import jwt from "jsonwebtoken";

const app = express();

app.use(express.json());

app.post("/login", (req, res) => {
  const { email, senha } = req.body;

  if (email !== "ana@email.com" || senha !== "1234") {
    return res.status(401).json({ erro: "Login inválido" });
  }

  const token = jwt.sign({usuarioId: 1,nome: "Ana"}, "minha-chave-secreta");

  res.json({ token });
});

app.listen(3000);
login válido
→
cria JWT
→
retorna token
O JWT carrega informações do usuário, como usuarioId, dentro do próprio token.
Rota protegida · 8 min

Agora usamos o token

app.get("/perfil", (req, res) => {
  const authorization = req.headers.authorization;

  if (!authorization) {
    return res.status(401).json({ erro: "Token não informado" });
  }

  const token = authorization.split(" ")[1];

  try {
    const usuario = jwt.verify(token, "minha-chave-secreta");
    res.json({ id: usuario.usuarioId, nome: usuario.nome });
  } catch {
    res.status(401).json({ erro: "Token inválido" });
  }
});
GET /perfil
→
envia JWT
→
jwt.verify()
→
retorna perfil
O cliente envia o token em Authorization: Bearer TOKEN
